-- 1. TIPOS / DOMÍNIOS
-- ============================================================

do $$
begin
  if not exists (
    select 1 from pg_type
    where typname = 'levenote_role'
  ) then
    create type public.levenote_role as enum ('ADM', 'MOD', 'USR');
  end if;
end
$$;


-- ============================================================
-- 2. FUNÇÃO GENÉRICA updated_at
-- ============================================================

create or replace function public.set_updated_at()
returns trigger
language plpgsql
security invoker
set search_path = public
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;


-- ============================================================
-- 3. PROFILES
-- Usada diretamente pelo HTML.
-- ============================================================

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  apelido text not null,
  emoji text not null default '🪶',
  apelido_cor text not null default '#d4889c',
  role public.levenote_role not null default 'USR',

  -- Silenciamento individual do chat.
  silenced_until timestamptz null,

  -- Controle do ingresso na comunidade.
  comunidade_entrou_em timestamptz not null default now(),

  -- Ticket permanente: sempre 1, nunca acumula.
  bilhete_permanente boolean not null default true,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint profiles_apelido_tamanho
    check (char_length(apelido) between 2 and 30),

  constraint profiles_apelido_formato
    check (apelido ~ '^[A-Za-zÀ-ÿ0-9 _-]+$'),

  constraint profiles_cor_hex
    check (apelido_cor ~ '^#[0-9A-Fa-f]{6}$')
);

create unique index if not exists profiles_apelido_lower_unique
  on public.profiles (lower(apelido));

create index if not exists profiles_role_idx
  on public.profiles(role);

create index if not exists profiles_silenced_idx
  on public.profiles(silenced_until);

drop trigger if exists trg_profiles_updated_at on public.profiles;

create trigger trg_profiles_updated_at
before update on public.profiles
for each row execute function public.set_updated_at();


-- ============================================================
-- 4. FUNÇÕES DE AUTORIZAÇÃO
-- Estas funções evitam consultar profiles diretamente dentro
-- das policies e reduzem problemas de recursão de RLS.
-- ============================================================

create or replace function public.current_user_role()
returns public.levenote_role
language sql
stable
security definer
set search_path = public
as $$
  select role
  from public.profiles
  where id = auth.uid()
  limit 1;
$$;

revoke all on function public.current_user_role() from public;
grant execute on function public.current_user_role() to anon, authenticated;

create or replace function public.is_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select coalesce(
    (select role = 'ADM'::public.levenote_role
     from public.profiles
     where id = auth.uid()
     limit 1),
    false
  );
$$;

revoke all on function public.is_admin() from public;
grant execute on function public.is_admin() to anon, authenticated;

create or replace function public.is_moderator()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select coalesce(
    (select role in (
       'ADM'::public.levenote_role,
       'MOD'::public.levenote_role
     )
     from public.profiles
     where id = auth.uid()
     limit 1),
    false
  );
$$;

revoke all on function public.is_moderator() from public;
grant execute on function public.is_moderator() to anon, authenticated;

-- ============================================================
-- 8. FUNÇÃO PARA VALIDAR O ENVIO DE MENSAGEM
-- Impede:
--   - chat geral desligado
--   - usuário silenciado
--   - mensagem > 200
-- Também preenche os dados visuais do perfil.
-- ============================================================

create or replace function public.prepare_chat_message()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  p public.profiles%rowtype;
  chat_ligado boolean;
begin
  if new.user_id is null then
    new.user_id := auth.uid();
  end if;

  if new.user_id <> auth.uid() and not public.is_admin() then
    raise exception 'Você não pode enviar mensagem em nome de outro usuário.';
  end if;

  select chat_ativo
    into chat_ligado
  from public.chat_settings
  where id = 1;

  if coalesce(chat_ligado, true) = false then
    raise exception 'O chat está silenciado.';
  end if;

  select *
    into p
  from public.profiles
  where id = new.user_id;

  if p.id is null then
    raise exception 'Perfil não encontrado.';
  end if;

  if p.silenced_until is not null
     and p.silenced_until > now() then
    raise exception 'Você está silenciado.';
  end if;

  if char_length(trim(new.mensagem)) < 1
     or char_length(new.mensagem) > 200 then
    raise exception 'A mensagem deve ter entre 1 e 200 caracteres.';
  end if;

  new.apelido := p.apelido;
  new.emoji := p.emoji;
  new.apelido_cor := p.apelido_cor;
  new.hidden := coalesce(new.hidden, false);
  new.deleted := coalesce(new.deleted, false);

  return new;
end;
$$;

drop trigger if exists trg_prepare_chat_message
on public.chat_messages;

create trigger trg_prepare_chat_message
before insert on public.chat_messages
for each row execute function public.prepare_chat_message();

-- 15. FUNÇÃO: GARANTIR BILHETE PERMANENTE

create or replace function public.garantir_bilhete_permanente()
returns trigger

as $$
begin
  insert into public.bilhetes (user_id, permanente, desafios, mes_referencia)

-- ============================================================
-- 16. FUNÇÃO: GARANTIR MÊS CORRETO
-- Se o usuário entrar em um novo mês, somente desafios zeram.
-- O permanente continua 1.
-- ============================================================

create or replace function public.normalizar_bilhetes_mes(p_user_id uuid)
returns public.bilhetes
language plpgsql
security definer
set search_path = public
as $$
declare
  r public.bilhetes;
  mes_atual date := date_trunc('month', current_date)::date;
begin
  insert into public.bilhetes(user_id, permanente, desafios, mes_referencia)
  values(p_user_id, 1, 0, mes_atual)
  on conflict(user_id) do nothing;

  select *
    into r
  from public.bilhetes
  where user_id = p_user_id
  for update;

  if r.mes_referencia <> mes_atual then
    update public.bilhetes
    set desafios = 0,
        permanente = 1,
        mes_referencia = mes_atual,
        updated_at = now()
    where user_id = p_user_id
    returning * into r;
  end if;

  return r;
end;
$$;

revoke all on function public.normalizar_bilhetes_mes(uuid) from public;
grant execute on function public.normalizar_bilhetes_mes(uuid) to anon, authenticated;


-- ============================================================
-- 17. FUNÇÃO ADMIN: ADICIONAR/REMOVER BILHETES
-- ============================================================

create or replace function public.admin_ajustar_bilhetes(
  p_user_id uuid,
  p_quantidade integer,
  p_observacao text default null
)
returns public.bilhetes
language plpgsql
security definer
set search_path = public
as $$
declare
  r public.bilhetes;
begin
  if not public.is_admin() then
    raise exception 'Somente ADM pode alterar bilhetes.';
  end if;

  if p_quantidade = 0 then
    raise exception 'A quantidade não pode ser zero.';
  end if;

  r := public.normalizar_bilhetes_mes(p_user_id);

  if r.desafios + p_quantidade < 0 then
    raise exception 'O usuário não pode ficar com bilhetes de desafio negativos.';
  end if;

  update public.bilhetes
  set desafios = desafios + p_quantidade,
      permanente = 1,
      updated_at = now()
  where user_id = p_user_id
  returning * into r;

  insert into public.bilhetes_movimentos(
    user_id,
    tipo,
    quantidade,
    origem,
    observacao,
    criado_por
  )
  values(
    p_user_id,
    case when p_quantidade > 0 then 'entrada' else 'saida' end,
    p_quantidade,
    'manual',
    p_observacao,
    auth.uid()
  );

  return r;
end;
$$;

revoke all on function public.admin_ajustar_bilhetes(uuid, integer, text)
from public;

grant execute on function public.admin_ajustar_bilhetes(uuid, integer, text)
to authenticated;


-- ============================================================
-- 18. FUNÇÃO ADMIN: RESETAR BILHETES DE DESAFIO
-- ============================================================

create or replace function public.admin_resetar_bilhetes(
  p_user_id uuid default null
)
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  qtd integer;
begin
  if not public.is_admin() then
    raise exception 'Somente ADM pode resetar bilhetes.';
  end if;

  if p_user_id is null then

    insert into public.bilhetes_movimentos(
      user_id, tipo, quantidade, origem, observacao, criado_por
    )
    select
      user_id,
      'reset_mensal',
      -desafios,
      'reset',
      'Reset geral de bilhetes de desafio',
      auth.uid()
    from public.bilhetes
    where desafios <> 0;

    update public.bilhetes
    set desafios = 0,
        permanente = 1,
        mes_referencia = date_trunc('month', current_date)::date,
        updated_at = now()
    where desafios <> 0;

    get diagnostics qtd = row_count;

    return qtd;

  else

    select desafios into qtd
    from public.bilhetes
    where user_id = p_user_id;

    if qtd is null then
      insert into public.bilhetes(user_id, permanente, desafios)
      values(p_user_id, 1, 0);
      return 0;
    end if;

    insert into public.bilhetes_movimentos(
      user_id, tipo, quantidade, origem, observacao, criado_por
    )
    values(
      p_user_id,
      'reset_mensal',
      -qtd,
      'reset',
      'Reset de bilhetes de desafio',
      auth.uid()
    );

    update public.bilhetes
    set desafios = 0,
        permanente = 1,
        mes_referencia = date_trunc('month', current_date)::date,
        updated_at = now()
    where user_id = p_user_id;

    return 1;
  end if;
end;
$$;

revoke all on function public.admin_resetar_bilhetes(uuid)
from public;

grant execute on function public.admin_resetar_bilhetes(uuid)
to authenticated;


-- ============================================================
-- 19. FUNÇÃO: TOTAL DE BILHETES DO USUÁRIO
-- permanente + desafios
-- ============================================================

create or replace function public.meus_bilhetes()
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  r public.bilhetes;
begin
  if auth.uid() is null then
    return 0;
  end if;

  r := public.normalizar_bilhetes_mes(auth.uid());

  return r.permanente + r.desafios;
end;
$$;

revoke all on function public.meus_bilhetes() from public;
grant execute on function public.meus_bilhetes() to anon, authenticated;

on public.profiles
for insert
to anon, authenticated
with check (
  id = auth.uid()
  and role = 'USR'::public.levenote_role
);

create policy profiles_update_self
on public.profiles
for update
to anon, authenticated
using (id = auth.uid() or public.is_admin())
with check (
  public.is_admin()
  or (
    id = auth.uid()
    and role = (
      select p.role
      from public.profiles p
      where p.id = auth.uid()
      limit 1
    )
  )
);

-- MOD também pode consultar perfis (já existe SELECT público), mas
-- somente ADM/MOD pode executar as operações de moderação abaixo.
create or replace function public.mod_silenciar_usuario(
  p_user_id uuid,
  p_minutos integer
)
returns timestamptz
language plpgsql
security definer
set search_path = public
as $$
declare
  v_until timestamptz;
  v_target_role public.levenote_role;
begin
  if not public.is_moderator() then
    raise exception 'Somente MOD ou ADM pode moderar o chat.';
  end if;

  select role into v_target_role
  from public.profiles
  where id = p_user_id;

  if v_target_role is null then
    raise exception 'Usuário não encontrado.';
  end if;

  if v_target_role = 'ADM'::public.levenote_role then
    raise exception 'Um MOD/ADM não pode silenciar um ADM.';
  end if;

  if p_minutos < 0 or p_minutos > 10080 then
    raise exception 'Prazo inválido. Use de 0 a 10080 minutos.';
  end if;

  v_until := case
    when p_minutos = 0 then null
    else now() + make_interval(mins => p_minutos)
  end;

  update public.profiles
  set silenced_until = v_until,
      updated_at = now()
  where id = p_user_id;

  return v_until;
end;
$$;

revoke all on function public.mod_silenciar_usuario(uuid, integer) from public;
grant execute on function public.mod_silenciar_usuario(uuid, integer) to authenticated;

-- ============================================================
-- DESAFIO: resposta do usuário + bilhete
-- A operação é atômica e não permite manipulação de bilhetes pelo
-- navegador. O usuário só informa a resposta.
-- ============================================================

create or replace function public.responder_desafio(
  p_desafio_id uuid,
  p_resposta text
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  d public.desafios;
  p public.profiles;
  v_correta boolean;
  v_primeiro_vencedor boolean := false;
  v_bilhetes public.bilhetes;
begin
  if auth.uid() is null then
    raise exception 'Usuário não autenticado.';
  end if;

  select * into d
  from public.desafios
  where id = p_desafio_id
    and ativo = true
  for update;

  if d.id is null then
    raise exception 'Este desafio não está ativo.';
  end if;

  select * into p
  from public.profiles
  where id = auth.uid();

  if p.id is null then
    raise exception 'Perfil não encontrado.';
  end if;

  if exists (
    select 1 from public.desafio_respostas
    where desafio_id = p_desafio_id
      and user_id = auth.uid()
  ) then
    raise exception 'Você já respondeu este desafio.';
  end if;

  v_correta := d.resposta_correta is not null
    and lower(trim(p_resposta)) = lower(trim(d.resposta_correta));

  insert into public.desafio_respostas(
    desafio_id,
    user_id,
    apelido,
    resposta,
    correta,
    bilhete_gerado
  )
  values(
    p_desafio_id,
    auth.uid(),
    p.apelido,
    left(trim(p_resposta), 500),
    v_correta,
    false
  );

  if v_correta then
    insert into public.bilhetes(
      user_id,
      permanente,
      desafios,
      mes_referencia
    )
    values(
      auth.uid(),
      1,
      0,
      date_trunc('month', current_date)::date
    )
    on conflict(user_id) do nothing;

    v_bilhetes := public.normalizar_bilhetes_mes(auth.uid());

    update public.bilhetes
    set desafios = desafios + 1,
        permanente = 1,
        updated_at = now()
    where user_id = auth.uid()
    returning * into v_bilhetes;

    update public.desafio_respostas
    set bilhete_gerado = true
    where desafio_id = p_desafio_id
      and user_id = auth.uid();

    insert into public.bilhetes_movimentos(
      user_id,
      tipo,
      quantidade,
      origem,
      desafio_id,
      observacao,
      criado_por
    )
    values(
      auth.uid(),
      'entrada',
      1,
      'desafio',
      p_desafio_id,
      'Resposta correta no desafio',
      auth.uid()
    );

    -- O primeiro acerto encerra o desafio e registra o vencedor.
    if d.vencedor_user_id is null then
      update public.desafios
      set vencedor_user_id = auth.uid(),
          vencedor_apelido = p.apelido,
          ativo = false,
          encerrado = true,
          encerrado_em = now(),
          atualizado_por = auth.uid(),
          updated_at = now()
      where id = p_desafio_id
        and vencedor_user_id is null;

      v_primeiro_vencedor := found;
    end if;
  end if;

  return jsonb_build_object(
    'correta', v_correta,
    'vencedor', v_primeiro_vencedor,
    'apelido', case when v_correta then p.apelido else null end,
    'bilhetes', case when v_correta then (v_bilhetes.permanente + v_bilhetes.desafios) else null end
  );
end;
$$;

revoke all on function public.responder_desafio(uuid, text) from public;
grant execute on function public.responder_desafio(uuid, text) to anon, authenticated;

create or replace function public.registrar_indicacao(
  p_indicador_user_id uuid default null,
  p_indicador_apelido text default null,
  p_resposta text default 'INFORMADO'
)
returns public.indicacoes
language plpgsql
security definer
set search_path=public
as $$
declare
  v_row public.indicacoes;
  v_indicado_apelido text;
  v_indicador_apelido text;
begin
  if auth.uid() is null then raise exception 'Usuário não autenticado.'; end if;
  if p_indicador_user_id=auth.uid() then raise exception 'Você não pode indicar a si mesmo.'; end if;

  select apelido into v_indicado_apelido from public.profiles where id=auth.uid();

  if p_indicador_user_id is not null then
    select apelido into v_indicador_apelido from public.profiles where id=p_indicador_user_id;
    if v_indicador_apelido is null then
      raise exception 'Usuário indicado não encontrado.';
    end if;
  else
    v_indicador_apelido=trim(coalesce(p_indicador_apelido,''));
    if v_indicador_apelido='' then raise exception 'Informe o apelido da pessoa que indicou você.'; end if;
  end if;

  insert into public.indicacoes(
    indicado_user_id, indicado_apelido,
    indicador_user_id, indicador_apelido,
    resposta, status, bilhete_atribuido
  ) values (
    auth.uid(), v_indicado_apelido,
    p_indicador_user_id, v_indicador_apelido,
    coalesce(p_resposta,'INFORMADO'), 'PENDENTE', false
  )
  on conflict(indicado_user_id) do update set
    indicador_user_id=excluded.indicador_user_id,
    indicador_apelido=excluded.indicador_apelido,
    resposta=excluded.resposta,
    status=case when public.indicacoes.status='ATRIBUIDO' then public.indicacoes.status else 'PENDENTE' end
  returning * into v_row;

  return v_row;
end;
$$;

revoke all on function public.registrar_indicacao(uuid,text,text) from public;
grant execute on function public.registrar_indicacao(uuid,text,text) to authenticated;

create or replace function public.levenote_stop_normalizar(p_texto text)
returns text language sql immutable
as $$
  select regexp_replace(
    translate(lower(trim(coalesce(p_texto,''))),
      'áàãâäéèêëíìîïóòõôöúùûüçñ',
      'aaaaaeeeeiiiiooooouuuucn'),
    '\\s+', ' ', 'g');
$$;

create or replace function public.admin_criar_stop(p_tema text,p_letra text,p_palavras text[])
returns uuid language plpgsql security definer set search_path=public
as $$
declare v_id uuid;
begin
  if coalesce((select role from public.profiles where id=auth.uid()),'USR') not in ('ADM','MOD') then raise exception 'Apenas ADM ou MOD pode criar desafios.'; end if;
  if coalesce(array_length(p_palavras,1),0)=0 then raise exception 'Informe pelo menos uma palavra.'; end if;
  insert into public.stop_desafios(tema,letra,criado_por)
  values(trim(p_tema),upper(left(trim(p_letra),1)),auth.uid()) returning id into v_id;
  insert into public.stop_palavras(desafio_id,palavra)
  select v_id,trim(x) from unnest(p_palavras) x where trim(x)<>'';
  return v_id;
end;
$$;

create or replace function public.admin_atualizar_stop(p_desafio_id uuid,p_tema text,p_letra text,p_palavras text[])
returns void language plpgsql security definer set search_path=public
as $$
begin
  if coalesce((select role from public.profiles where id=auth.uid()),'USR') not in ('ADM','MOD') then raise exception 'Apenas ADM ou MOD pode editar desafios.'; end if;
  update public.stop_desafios set tema=trim(p_tema),letra=upper(left(trim(p_letra),1)),encerrado=false where id=p_desafio_id;
  if not found then raise exception 'STOP não encontrado.'; end if;
  delete from public.stop_palavras where desafio_id=p_desafio_id;
  insert into public.stop_palavras(desafio_id,palavra)
  select p_desafio_id,trim(x) from unnest(p_palavras) x where trim(x)<>'';
end;
$$;

create or replace function public.responder_stop(p_desafio_id uuid,p_resposta text)
returns jsonb language plpgsql security definer set search_path=public
as $$
declare
  v_palavra public.stop_palavras%rowtype;
  v_apelido text;
  v_restantes integer;
begin
  if auth.uid() is null then raise exception 'Usuário não autenticado.'; end if;
  select * into v_palavra from public.stop_palavras
  where desafio_id=p_desafio_id and usada=false
    and public.levenote_stop_normalizar(palavra)=public.levenote_stop_normalizar(p_resposta)
  order by created_at limit 1 for update;
  if not found then return jsonb_build_object('correta',false,'mensagem','Essa palavra não está disponível.'); end if;
  select apelido into v_apelido from public.profiles where id=auth.uid();
  update public.stop_palavras set usada=true,acertada_por=auth.uid(),acertada_por_apelido=v_apelido,usada_em=now() where id=v_palavra.id and usada=false;
  if not found then return jsonb_build_object('correta',false,'mensagem','Outra pessoa acabou de acertar essa palavra.'); end if;
  insert into public.bilhetes(user_id,permanente,desafios) values(auth.uid(),1,1)
  on conflict(user_id) do update set desafios=coalesce(public.bilhetes.desafios,0)+1;
  select count(*)::integer into v_restantes from public.stop_palavras where desafio_id=p_desafio_id and usada=false;
  if v_restantes=0 then update public.stop_desafios set ativo=false,encerrado=true,encerrado_em=now() where id=p_desafio_id and ativo=true; end if;
  return jsonb_build_object('correta',true,'mensagem','Palavra correta! Você ganhou 1 bilhete.','palavra',v_palavra.palavra,'acertado_por',v_apelido,'restantes',v_restantes);
end;
$$;

-- ---------------------------------------------------------------
create table if not exists public.bilhetes (
  user_id uuid primary key references public.profiles(id) on delete cascade,
  permanente integer not null default 1,
  desafios integer not null default 0,
  updated_at timestamptz not null default now(),
  constraint bilhetes_permanente_exatamente_1 check (permanente = 1),
  constraint bilhetes_desafios_nao_negativos check (desafios >= 0)
);

alter table public.bilhetes add column if not exists permanente integer;
alter table public.bilhetes add column if not exists desafios integer;
alter table public.bilhetes add column if not exists updated_at timestamptz;

update public.bilhetes
set permanente = 1
where permanente is distinct from 1;

update public.bilhetes
set desafios = 0
where desafios is null or desafios < 0;

update public.bilhetes
set updated_at = coalesce(updated_at, now())
where updated_at is null;

alter table public.bilhetes alter column permanente set default 1;
alter table public.bilhetes alter column desafios set default 0;
alter table public.bilhetes alter column updated_at set default now();
alter table public.bilhetes alter column permanente set not null;
alter table public.bilhetes alter column desafios set not null;
alter table public.bilhetes alter column updated_at set not null;

-- Garante que todo perfil existente tenha o 1 permanente.
insert into public.bilhetes (user_id, permanente, desafios)
select p.id, 1, 0
from public.profiles p
left join public.bilhetes b on b.user_id = p.id
where b.user_id is null
on conflict (user_id) do nothing;

$$;

create or replace function public.admin_criar_stop(
  p_tema text,
  p_letra text,
  p_palavras text[]
)
returns uuid
language plpgsql
security definer
set search_path = public
as $$
declare
  v_id uuid;
  v_palavra text;
begin
  if not public.levenote_is_mod() then
    raise exception 'Apenas ADM ou MOD pode criar STOP.' using errcode = '42501';
  end if;
  if nullif(trim(p_tema), '') is null then raise exception 'Tema obrigatório.'; end if;
  if nullif(trim(p_letra), '') is null then raise exception 'Letra obrigatória.'; end if;
  if coalesce(array_length(p_palavras,1),0) = 0 then raise exception 'Cadastre pelo menos uma palavra.'; end if;

  insert into public.stop_desafios (tema, letra, ativo, encerrado, created_by)
  values (trim(p_tema), upper(left(trim(p_letra),1)), false, false, auth.uid())
  returning id into v_id;

  foreach v_palavra in array p_palavras loop
    if nullif(trim(v_palavra),'') is not null then
      insert into public.stop_palavras (desafio_id, palavra)
      values (v_id, trim(v_palavra));
    end if;
  end loop;

  return v_id;
end;
$$;

-- A resposta é atômica: a palavra é marcada como usada e o bilhete é
-- incrementado na mesma transação. Duas pessoas não conseguem ganhar pelo
-- mesmo registro ao mesmo tempo.
create or replace function public.responder_stop(
  p_desafio_id uuid,
  p_resposta text
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_palavra public.stop_palavras;
  v_resposta text := lower(trim(coalesce(p_resposta,'')));
  v_total integer;
begin
  if auth.uid() is null then
    raise exception 'Usuário não autenticado.' using errcode='42501';
  end if;

  if not exists (select 1 from public.stop_desafios where id=p_desafio_id and ativo=true) then
    return jsonb_build_object('correta',false,'mensagem','Este STOP não está ativo.');
  end if;

  select sp.* into v_palavra
  from public.stop_palavras sp
  where sp.desafio_id=p_desafio_id
    and coalesce(sp.usada,false)=false
    and lower(trim(sp.palavra))=v_resposta
  order by sp.created_at
  for update skip locked
  limit 1;

  if not found then
    return jsonb_build_object('correta',false,'mensagem','Essa palavra não está disponível.');
  end if;

  update public.stop_palavras
  set usada=true, respondido_por=auth.uid(), respondido_em=now()
  where id=v_palavra.id;

  insert into public.bilhetes (user_id, permanente, desafios)
  values (auth.uid(),1,1)
  on conflict (user_id) do update
    set permanente=1,
        desafios=public.bilhetes.desafios+1,
        updated_at=now();

  select permanente+desafios into v_total
  from public.bilhetes where user_id=auth.uid();

  -- Se esta foi a última palavra disponível, o STOP é encerrado
  -- automaticamente. Assim nenhum usuário depende de uma ação manual
  -- do ADM/MOD para voltar à tela de espera.
  if not exists (
    select 1
    from public.stop_palavras
    where desafio_id=p_desafio_id
      and coalesce(usada,false)=false
  ) then
    update public.stop_desafios
    set ativo=false,
        encerrado=true,
        encerrado_em=now()
    where id=p_desafio_id and ativo=true;

    return jsonb_build_object(
      'correta',true,
      'encerrado',true,
      'mensagem','Palavra correta! Você ganhou 1 bilhete. STOP encerrado.',
      'total_bilhetes',v_total
    );
  end if;

  return jsonb_build_object(
    'correta',true,
    'encerrado',false,
    'mensagem','Palavra correta! Você ganhou 1 bilhete.',
    'total_bilhetes',v_total
  );
end;
$$;

grant execute on function public.responder_stop(uuid,text) to authenticated;

create or replace function public.roleta_selecionar_proximo(p_sorteio_id uuid)
returns public.roleta_sorteios
language plpgsql
security definer
set search_path = public
as $$
declare
  v public.roleta_sorteios;
  n jsonb;
  v_total bigint := 0;
  v_pick bigint;
  v_acc bigint := 0;
  v_winner_id uuid;
  v_winner_name text;
  v_winner_emoji text;
  v_winner_total integer;
begin
  if not public.levenote_is_admin() then
    raise exception 'Apenas ADM pode sortear.' using errcode='42501';
  end if;

  select * into v
  from public.roleta_sorteios
  where id=p_sorteio_id
  for update;

  if not found then raise exception 'Sorteio não encontrado.'; end if;
  if v.status <> 'ATIVO' then raise exception 'Sorteio não está ativo.'; end if;
  if not coalesce(v.exibida,false) then raise exception 'Lance o sorteio no Chat antes de sortear.'; end if;
  if v.vencedor_apelido is not null then raise exception 'Este sorteio já foi realizado.'; end if;

  for n in select value from jsonb_array_elements(coalesce(v.participantes_nomes,'[]'::jsonb)) loop
    if coalesce((n->>'id')::uuid, null) is not null then
      v_total := v_total + greatest(1, coalesce((n->>'total')::integer,1));
    end if;
  end loop;

  if v_total <= 0 then raise exception 'Não há participantes com bilhetes.'; end if;

  v_pick := floor(random()*v_total)::bigint + 1;
  v_acc := 0;

  for n in select value from jsonb_array_elements(coalesce(v.participantes_nomes,'[]'::jsonb)) loop
    if coalesce((n->>'id')::uuid, null) is not null then
      v_acc := v_acc + greatest(1, coalesce((n->>'total')::integer,1));
      if v_acc >= v_pick then
        v_winner_id := (n->>'id')::uuid;
        v_winner_name := n->>'apelido';
        v_winner_emoji := coalesce(n->>'emoji','👤');
        v_winner_total := greatest(1, coalesce((n->>'total')::integer,1));
        exit;
      end if;
    end if;
  end loop;

  update public.roleta_sorteios
  set ultimo_selecionado_apelido=v_winner_name,
      vencedor_apelido=v_winner_name,
      giro_seq=coalesce(giro_seq,0)+1,
      updated_at=now()
  where id=p_sorteio_id
  returning * into v;

  return v;
end;
$$;

grant execute on function public.roleta_selecionar_proximo(uuid) to authenticated;

revoke all on function public.get_community_limit() from public;
grant execute on function public.get_community_limit() to authenticated;

revoke all on function public.get_community_usage() from public;
grant execute on function public.get_community_usage() to authenticated;

revoke all on function public.admin_configurar_limite_comunidade(integer) from public;
grant execute on function public.admin_configurar_limite_comunidade(integer) to authenticated;

-- Bloqueio real no banco. O lock transacional evita que dois cadastros
-- simultâneos ultrapassem o limite por uma condição de corrida.
create or replace function public.enforce_community_user_limit()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_limite integer;
  v_atuais integer;
begin
  -- O perfil já existente não conta novamente.
  if exists (select 1 from public.profiles where id = new.id) then
    return new;
  end if;

  -- Um lock único para serializar novas entradas na comunidade.
  perform pg_advisory_xact_lock(846271);

  select user_limit into v_limite
  from public.community_settings
  where id=1;

  v_limite := coalesce(v_limite,50);

  select count(*) into v_atuais
  from public.profiles;

  if v_atuais >= v_limite then
    raise exception 'limite_comunidade_atingido';
  end if;

  return new;
end;
$$;

drop trigger if exists trg_enforce_community_user_limit on public.profiles;
create trigger trg_enforce_community_user_limit
before insert on public.profiles
for each row execute function public.enforce_community_user_limit();

-- ================================================================
-- BLOCO E — NOTIFICAÇÕES PÚBLICAS + DESTINATÁRIO
-- ================================================================
-- Avisos destinados a TODOS usam destinatario_id = NULL.
-- O usuário pode ler esses avisos, mas não pode alterar/excluir o registro
-- público diretamente; o status individual acima é usado pelo HTML.

drop policy if exists "notificacoes leitura do destinatario" on public.notificacoes;
create policy "notificacoes leitura do destinatario"
on public.notificacoes
for select to authenticated
using (destinatario_id is null or destinatario_id = auth.uid());

-- ================================================================
-- BLOCO F — REALTIME NECESSÁRIO AO APP
-- ================================================================
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname='supabase_realtime'
      and schemaname='public'
      and tablename='roleta_sorteios'
  ) then
    execute 'alter publication supabase_realtime add table public.roleta_sorteios';
  end if;
exception when others then
  raise notice 'Realtime da roleta não pôde ser adicionado automaticamente: %', sqlerrm;
end $$;

-- ================================================================
-- BLOCO G — GARANTIA DE BILHETE PERMANENTE PARA OS PERFIS EXISTENTES
-- ================================================================
insert into public.bilhetes (user_id, permanente, desafios)
select p.id, 1, 0
from public.profiles p
left join public.bilhetes b on b.user_id = p.id
where b.user_id is null
on conflict (user_id) do nothing;

-- ================================================================
-- BLOCO H — DEFINIÇÃO DO ADM
-- ================================================================
-- NÃO preencha automaticamente um UUID desconhecido.
-- Depois de descobrir o UUID do seu usuário em Authentication > Users,
-- execute, por exemplo:
--
-- update public.profiles
-- set role='ADM'
-- where id='COLE-SEU-UUID-AQUI';
--
-- Para MOD:
-- update public.profiles set role='MOD' where id='COLE-SEU-UUID-AQUI';
--
-- ================================================================
-- BLOCO I — VERIFICAÇÃO FINAL
-- ================================================================
select p.id, p.apelido, p.emoji, p.role,
       coalesce(b.permanente,1) as permanente,
       coalesce(b.desafios,0) as desafios,
       coalesce(b.permanente,1)+coalesce(b.desafios,0) as total
from public.profiles p
left join public.bilhetes b on b.user_id=p.id
order by p.apelido;

select id, status, exibida, vencedor_apelido, giro_seq, created_at
from public.roleta_sorteios
order by created_at desc
limit 20;

-- Verificação do limite da comunidade. Deve retornar 50 após a instalação.
select * from public.community_settings where id=1;
select public.get_community_usage() as usuarios_atuais, public.get_community_limit() as limite_comunidade;

commit;
